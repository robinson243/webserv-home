#CGI with GET

curl -v http://127.0.0.1:8080/cgi-bin/echo.py
curl -v http://127.0.0.1:8080/cgi-bin/env.py

#CGI with POST

curl -v -X POST http://127.0.0.1:8080/cgi-bin/echo.py \
  -H "Content-Type: application/x-www-form-urlencoded" \
  --data "message=hello&user=rob"

curl -v -X POST http://127.0.0.1:8080/cgi-bin/env.py \
  -H "Content-Type: application/x-www-form-urlencoded" \
  --data "a=1&b=2"
